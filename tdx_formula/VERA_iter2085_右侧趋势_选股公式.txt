{── 条件1: 收盘价创 12 日新高 (与前12日最高收盘比, 不含当日) ──}
COND1 := CLOSE >= REF(HHV(CLOSE, 12), 1);

{── 条件2: 今天刚上穿 5 日均线 (昨天还在线下, 必须"刚刚"突破) ──}
COND2 := CROSS(CLOSE, MA(CLOSE, 5));

{── 条件3: 均线多头排列 5日>10日>20日 (楼梯层层向上, 趋势已形成) ──}
COND3 := MA(CLOSE, 5) > MA(CLOSE, 10) AND MA(CLOSE, 10) > MA(CLOSE, 20);

{── 输出: 三条件同时满足 ──}
XG: COND1 AND COND2 AND COND3;
